  hosts: SX # dove X è il numero specifico del server

- name: 
  hosts: S*
  become: true
  roles:
    - ldap_utils        # interrogazione in lettura/scrittura della directory
    - ldap_creds        # credenziali di bind in /etc/ldap.admin.cred
    - rsyslog           # ricezione remota di logs
    - create_dir        # /var/spool/printjobs/pending
    - snmp_agent        # espone la lunghezza della coda
    - sudoers           # non posso dar per scontato che Debian-snmp abbia accesso a un determinato file o cartella
    - ssh_keys          # root si connette all'altro server (se richiesto e presente più di uno)
    - ssh_authkeys      # root e' bersaglio della connessione dall'altro
    - at_atd            # pianificazione della copia di ridondanza
    - install_script_server # per installazione di uno script dentro al server
    - systemd_service   # per far si che uno script sia avviato al boot e sempre riavviato
    - cron              # per lanciare uno script periodicamente
